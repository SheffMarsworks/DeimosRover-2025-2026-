Set up the project

In .bashrc, add
    source /opt/ros/humble/setup.bash
    source /usr/share/colcon_argcomplete/hook/colcon-argcomplete.bash

Install necessary dependencies
    sudo apt install -y python3-colcon-clean ros-humble-xacro ros-humble-joint-state-publisher-gui ros-humble-ros-gz ros-humble-ros-gz-bridge ros-humble-ros-gz-sim ros-humble-ros-gz-interfaces ros-humble-ign-ros2-control ros-humble-twist-stamper ros-humble-imu-tools ros-humble-ros2controlcli ros-humble-controller-manager ros-humble-ros2-controllers ros-humble-joy-teleop ros-humble-joy ros-humble-rtabmap-ros

    sudo apt install 

    cd code/marswork/DeimosRover-2025-2026-/
    colcon build --symlink-install
    . install/setup.bash

Visualize robot in Rviz
    ros2 launch rover_description rviz_demo.launch.py

Open marsworld in Gazebo
    ign gazebo -r $(ros2 pkg prefix rover_description)/share/rover_description/worlds/mars.world.sdf

Simulation
    ros2 launch rover_bringup simulation.launch.py
    ros2 launch rover_bringup simulation.launch.py world:=warehouse
    ros2 launch rover_bringup simulation.launch.py world:=mars

Teleop (default is keyboard)
    ros2 launch rover_bringup simulation.launch.py world:=warehouse mode:=teleop
    ros2 launch rover_bringup simulation.launch.py world:=warehouse mode:=teleop teleop:=joystick
Slam
    ros2 launch rover_bringup simulation.launch.py world:=warehouse mode:=slam
    ros2 launch rover_bringup simulation.launch.py world:=industrial mode:=slam Y:=1.5708 x:=1.35

(may need to run if transform node died)
    ros2 run tf2_ros static_transform_publisher 0 0 0 0 0 0 odom base_footprint
(may need to run if controller node died)
    ros2 launch rover_controller controller.launch.py
(debug transform tree)
    ros2 run tf2_tools view_frames
(clean workspace and rebuild and source)
    colcon clean workspace -y && colcon build --symlink-install && . install/setup.bash

    ign fuel download -t world -u "https://fuel.gazebosim.org/1.0/openrobotics/worlds/industrial-warehouse"
    world is saved in ~/.ignition/fuel/fuel.gazebosim.org/openrobotics/worlds/...
    can view by tree ~/.ignition/fuel/fuel.gazebosim.org/openrobotics/worlds
    or by explorer explorer.exe $(wslpath -w ~/.ignition/fuel/fuel.gazebosim.org/openrobotics/worlds)
